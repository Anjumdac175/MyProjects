1.Containerize a Plain Java Hello World Application

public class HelloWorld {
    public static void main(String[] args) {
        System.out.println("Hello World from Docker!");
    }
}

FROM openjdk:17
WORKDIR /app
COPY HelloWorld.java .
RUN javac HelloWorld.java
CMD ["java", "HelloWorld"]

docker build -t hello-java .
docker run hello-java

2. Java with MySQL Integration as discussed in the class

import java.sql.*;
public class JdbcTest {
    public static void main(String[] args) {
        try {
            Connection conn = DriverManager.getConnection(
                "jdbc:mysql://mysql:3306/testdb", "root", "password");
            Statement stmt = conn.createStatement();
            ResultSet rs = stmt.executeQuery("SELECT * FROM customer");
            while (rs.next())
                System.out.println(rs.getInt(1) + " " + rs.getString(2));
            conn.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}


FROM openjdk:17
WORKDIR /app
COPY . .
RUN javac JdbcTest.java
CMD ["java", "JdbcTest"]

14. Use a yaml file to run nginx deployment with 5 pods in Kubernetes.
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nginx-deployment
spec:
  replicas: 5
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
      - name: nginx
        image: nginx
        ports:
        - containerPort: 80

15. Write a Java program to accept two numbers from the user and display their sum. Create a Docker image and run it as a container.

import java.util.Scanner;
public class Add {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        System.out.println("Sum: " + (a + b));
    }
}

FROM openjdk:17
WORKDIR /app
COPY Add.java .
RUN javac Add.java
CMD ["java", "Add"]

docker build -t add-java .
docker run -it add-java
