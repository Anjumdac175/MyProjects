1.Consider following tables
    product(pid,pname,price,qty)
    product_details(pid,amt)
    amt =price * qty
Write a trigger which will calculate amt every time any update is done in the price of an item. 
-------------------------------------------------------------------------------------------------

Ans :

mysql> create table product
    -> (pid int primary key auto_increment,pname varchar(20) unique not null,price int,qty int);
Query OK, 0 rows affected (0.08 sec)

mysql> create table product_details
	-> (pid int primary key, amount int, 
	foreign key(pid) references product(pid) on delete cascade);
Query OK, 0 rows affected (0.05 sec)

mysql> delimiter ;
mysql> delimiter //
mysql> create trigger calculate_amt
    -> after update
    -> on product
    -> for each row
    -> begin
    ->  declare amount int;
    ->  if old.price <> new.price
    ->  then
    ->          update product_details
    ->          set amount = new.price * new.qty
    ->          where pid = new.pid;
    ->  end if;
    ->
    -> end //
Query OK, 0 rows affected (0.03 sec)

5.create table med_master
(mno int primary key auto_increment,
 mname varchar(15),
 stock int
);
create table med_order
(mid int primary key auto_increment,
 mno int,
 qty int);
Create above tables and write the trigger which will insert data in med_order table if stock drops down below 15.
Use update after trigger.
------------------------------------------------------------------------------------------------------------------
Ans:

mysql> delimiter //
mysql> create trigger update_stock
    -> after update
    -> on med_master
    -> for each row
    -> begin
    ->  declare qty int;
    ->  if new.stock < 15
    ->  then
    ->          set qty = new.stock;
    ->          insert into med_order(mno,qty)
    ->          values(new.mno,qty);
    ->  end if;
    -> end //
Query OK, 0 rows affected (0.01 sec)
