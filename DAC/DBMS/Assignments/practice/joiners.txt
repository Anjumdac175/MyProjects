4.Retrieve all employees and their all department names

mysql> select e.eid,e.ename,d.deptno,d.deptname
    -> from emp e
    -> left join dept d
    -> on e.deptno = d.deptno
    -> union
    -> select e.eid,e.ename,d.deptno,d.deptname
    -> from emp e
    -> right join dept d
    -> on e.deptno = d.deptno
    -> ;
